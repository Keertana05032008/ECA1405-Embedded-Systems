
#include <Arduino.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "freertos/semphr.h"

#define USE_MUTEX 1
#define RUN_CORE 1

SemaphoreHandle_t resourceLock;
uint32_t tStart;

#define NOW() (millis() - tStart)

void busyWork(uint32_t ms) {
  uint32_t start = millis();
  while (millis() - start < ms) {
  }
}

void LowTask(void *pv) {
  vTaskDelay(pdMS_TO_TICKS(100));

  xSemaphoreTake(resourceLock, portMAX_DELAY);

  Serial.printf("[%4lu ms] LOW -> resource LOCKED (priority %u)\n",
                (unsigned long)NOW(),
                (unsigned)uxTaskPriorityGet(NULL));

  for (int i = 1; i <= 10; i++) {
    busyWork(100);

    Serial.printf("[%4lu ms] LOW -> working %d/10, current priority %u\n",
                  (unsigned long)NOW(), i,
                  (unsigned)uxTaskPriorityGet(NULL));
  }

  xSemaphoreGive(resourceLock);

  Serial.printf("[%4lu ms] LOW -> resource RELEASED, priority now %u\n",
                (unsigned long)NOW(),
                (unsigned)uxTaskPriorityGet(NULL));

  vTaskDelete(NULL);
}

void MediumTask(void *pv) {
  vTaskDelay(pdMS_TO_TICKS(400));

  Serial.printf("[%4lu ms] MEDIUM -> starts 2000 ms of CPU work\n",
                (unsigned long)NOW());

  for (int i = 1; i <= 4; i++) {
    busyWork(500);

    Serial.printf("[%4lu ms] MEDIUM -> working %d/4\n",
                  (unsigned long)NOW(), i);
  }

  Serial.printf("[%4lu ms] MEDIUM -> finished\n",
                (unsigned long)NOW());

  vTaskDelete(NULL);
}

void HighTask(void *pv) {
  vTaskDelay(pdMS_TO_TICKS(300));

  Serial.printf("[%4lu ms] HIGH -> requests the resource\n",
                (unsigned long)NOW());

  uint32_t t0 = millis();

  xSemaphoreTake(resourceLock, portMAX_DELAY);

  uint32_t waited = millis() - t0;

  Serial.printf("[%4lu ms] HIGH -> resource LOCKED after waiting %lu ms\n",
                (unsigned long)NOW(), (unsigned long)waited);

  xSemaphoreGive(resourceLock);

  Serial.println("==============================================");

  Serial.printf("RESULT: HIGH waited %lu ms (%s)\n",
                (unsigned long)waited,
                USE_MUTEX ? "mutex, priority inheritance"
                          : "binary semaphore, NO inheritance");

  Serial.println("==============================================");

  vTaskDelete(NULL);
}

void setup() {
  Serial.begin(115200);
  delay(1000);

  Serial.println("==============================================");
  Serial.printf("PRIORITY INVERSION TEST (USE_MUTEX = %d)\n",
                USE_MUTEX);
  Serial.println("LOW = P1, MEDIUM = P2, HIGH = P3, core 1");
  Serial.println("==============================================");

  if (USE_MUTEX) {
    resourceLock = xSemaphoreCreateMutex();
  } else {
    resourceLock = xSemaphoreCreateBinary();

    if (resourceLock != NULL) {
      xSemaphoreGive(resourceLock);
    }
  }

  if (resourceLock == NULL) {
    Serial.println("Creation failed");
    while (1) delay(1000);
  }

  tStart = millis();

  xTaskCreatePinnedToCore(
    LowTask, "Low", 4096, NULL, 1, NULL, RUN_CORE
  );

  xTaskCreatePinnedToCore(
    MediumTask, "Medium", 4096, NULL, 2, NULL, RUN_CORE
  );

  xTaskCreatePinnedToCore(
    HighTask, "High", 4096, NULL, 3, NULL, RUN_CORE
  );
}

void loop() {
  vTaskDelay(pdMS_TO_TICKS(1000));
}
