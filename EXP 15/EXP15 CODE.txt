
#include <Arduino.h>

#define BUTTON_PIN 0
#define RUN_CORE 1

TaskHandle_t InterruptTaskHandle = NULL;
volatile uint32_t interruptCount = 0;

void IRAM_ATTR ButtonISR() {
  static uint32_t lastTime = 0;
  uint32_t now = millis();

  if (now - lastTime < 200) return;
  lastTime = now;

  interruptCount++;

  BaseType_t xHigherPriorityTaskWoken = pdFALSE;

  vTaskNotifyGiveFromISR(
    InterruptTaskHandle,
    &xHigherPriorityTaskWoken
  );

  if (xHigherPriorityTaskWoken) {
    portYIELD_FROM_ISR();
  }
}

void InterruptTask(void *parameter) {
  while (1) {
    uint32_t pending = ulTaskNotifyTake(
      pdTRUE, portMAX_DELAY
    );

    Serial.printf(
      "HIGH PRIORITY TASK -> Interrupt Count: %lu (notifications: %lu)\n",
      (unsigned long)interruptCount,
      (unsigned long)pending
    );

    Serial.println("Deferred processing executed.");
    Serial.println("------------------------------");
  }
}

void BackgroundTask(void *parameter) {
  while (1) {
    Serial.println(
      "LOW PRIORITY TASK -> Background processing"
    );

    vTaskDelay(pdMS_TO_TICKS(1000));
  }
}

void setup() {
  Serial.begin(115200);
  delay(1000);

  pinMode(BUTTON_PIN, INPUT_PULLUP);

  Serial.println("=================================");
  Serial.println("ISR PRIORITY AND DEFERRED PROCESSING");
  Serial.println("=================================");

  xTaskCreatePinnedToCore(
    InterruptTask,
    "InterruptTask",
    4096,
    NULL,
    3,
    &InterruptTaskHandle,
    RUN_CORE
  );

  xTaskCreatePinnedToCore(
    BackgroundTask,
    "BackgroundTask",
    4096,
    NULL,
    1,
    NULL,
    RUN_CORE
  );

  attachInterrupt(
    digitalPinToInterrupt(BUTTON_PIN),
    ButtonISR,
    FALLING
  );

  Serial.println("Interrupt configured on GPIO 0.");
  Serial.println("Interrupt Task Priority = 3");
  Serial.println("Background Task Priority = 1");
  Serial.println("Press the BOOT button.");
  Serial.println("=================================");
}

void loop() {
  vTaskDelay(pdMS_TO_TICKS(1000));
}
