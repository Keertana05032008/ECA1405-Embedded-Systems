
#include <Arduino.h>

SemaphoreHandle_t binarySemaphore;
SemaphoreHandle_t counterMutex;

int sharedCounter = 0;
volatile int tasksFinished = 0;

#define UPDATES 10

void BinaryProducerTask(void *pvParameters) {
  int eventNumber = 0;

  while (1) {
    eventNumber++;
    Serial.printf("[BINARY] Producer event %d\n", eventNumber);

    if (xSemaphoreGive(binarySemaphore) == pdTRUE)
      Serial.println("[BINARY] Signal sent");
    else
      Serial.println("[BINARY] Semaphore full");

    vTaskDelay(pdMS_TO_TICKS(2000));
  }
}

void BinaryConsumerTask(void *pvParameters) {
  while (1) {
    if (xSemaphoreTake(binarySemaphore, portMAX_DELAY) == pdTRUE)
      Serial.println("[BINARY] Consumer received signal");
  }
}

void CounterTask(void *pvParameters) {
  int id = (int)(intptr_t)pvParameters;

  for (int i = 0; i < UPDATES; i++) {
    xSemaphoreTake(counterMutex, portMAX_DELAY);

    int temp = sharedCounter;
    vTaskDelay(pdMS_TO_TICKS(10));
    sharedCounter = temp + 1;

    Serial.printf("[MUTEX] Task %d counter = %d\n",
                  id, sharedCounter);

    xSemaphoreGive(counterMutex);

    vTaskDelay(pdMS_TO_TICKS(id == 1 ? 50 : 70));
  }

  tasksFinished++;
  vTaskDelete(NULL);
}

void setup() {
  Serial.begin(115200);
  delay(1000);

  Serial.println("BINARY SEMAPHORE EXPERIMENT");

  binarySemaphore = xSemaphoreCreateBinary();
  counterMutex = xSemaphoreCreateMutex();

  if (binarySemaphore == NULL || counterMutex == NULL) {
    Serial.println("ERROR: Semaphore or mutex creation failed");
    while (1) vTaskDelay(pdMS_TO_TICKS(1000));
  }

  xTaskCreate(BinaryProducerTask, "BinProd", 4096, NULL, 2, NULL);
  xTaskCreate(BinaryConsumerTask, "BinCons", 4096, NULL, 1, NULL);

  vTaskDelay(pdMS_TO_TICKS(10000));

  Serial.println("\nMUTEX SHARED DATA PROTECTION EXPERIMENT");

  xTaskCreatePinnedToCore(
    CounterTask, "Counter1", 4096, (void*)1, 1, NULL, 1
  );

  xTaskCreatePinnedToCore(
    CounterTask, "Counter2", 4096, (void*)2, 1, NULL, 1
  );
}

void loop() {
  static bool reported = false;

  if (tasksFinished == 2 && !reported) {
    Serial.printf("FINAL sharedCounter = %d (expected %d)\n",
                  sharedCounter, 2 * UPDATES);
    reported = true;
  }

  vTaskDelay(pdMS_TO_TICKS(200));
}
