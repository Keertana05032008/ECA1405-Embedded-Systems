
#include <Arduino.h>
#include <freertos/FreeRTOS.h>
#include <freertos/task.h>
#include <freertos/portable.h>

#define BLOCK_SIZE 32
#define NUM_BLOCKS 4

// Fixed-size memory pool
static uint8_t pool[NUM_BLOCKS][BLOCK_SIZE];
static bool used[NUM_BLOCKS] = {false};

void *poolAlloc(void) {
  for (int i = 0; i < NUM_BLOCKS; i++) {
    if (!used[i]) {
      used[i] = true;
      return pool[i];
    }
  }
  return NULL;
}

void poolFree(void *p) {
  for (int i = 0; i < NUM_BLOCKS; i++) {
    if (p == pool[i]) {
      used[i] = false;
      return;
    }
  }
}

// Uses stack memory
void stackHungry(void) {
  volatile uint8_t buffer[1500];

  for (int i = 0; i < 1500; i++) {
    buffer[i] = (uint8_t)i;
  }

  Serial.printf("stackHungry(): buffer[1499] = %u\n",
                (unsigned)buffer[1499]);
}

void MemoryTask(void *pv) {
  // Part A: Heap allocation and release
  Serial.println("\n--- PART A: HEAP ALLOCATION ---");

  size_t before = ESP.getFreeHeap();
  int *data = (int *)pvPortMalloc(10 * sizeof(int));
  size_t after = ESP.getFreeHeap();

  if (data == NULL) {
    Serial.println("Allocation FAILED");
  } else {
    Serial.printf("Requested: %u bytes\n",
                  (unsigned)(10 * sizeof(int)));
    Serial.printf("Heap before: %u bytes\n", (unsigned)before);
    Serial.printf("Heap after: %u bytes\n", (unsigned)after);
    Serial.printf("Heap reduction: %u bytes\n",
                  (unsigned)(before - after));

    vPortFree(data);
    Serial.printf("Heap after release: %u bytes\n",
                  (unsigned)ESP.getFreeHeap());
  }

  // Part B: Memory leak demonstration
  Serial.println("\n--- PART B: MEMORY LEAK ---");

  void *leak[5] = {NULL, NULL, NULL, NULL, NULL};
  size_t start = ESP.getFreeHeap();

  for (int i = 0; i < 5; i++) {
    leak[i] = pvPortMalloc(1024);
    size_t nowFree = ESP.getFreeHeap();

    if (leak[i] != NULL) {
      Serial.printf("After allocation %d: free heap %u bytes (lost %u)\n",
                    i + 1, (unsigned)nowFree,
                    (unsigned)(start - nowFree));
    } else {
      Serial.printf("Allocation %d FAILED\n", i + 1);
    }
  }

  for (int i = 0; i < 5; i++) {
    if (leak[i] != NULL) {
      vPortFree(leak[i]);
    }
  }

  Serial.printf("After freeing all: %u bytes\n",
                (unsigned)ESP.getFreeHeap());
  Serial.printf("Lowest free heap ever: %u bytes\n",
                (unsigned)ESP.getMinFreeHeap());

  // Part C: Stack high-water mark
  Serial.println("\n--- PART C: STACK HIGH-WATER MARK ---");

  UBaseType_t hwm1 = uxTaskGetStackHighWaterMark(NULL);
  Serial.printf("Before stackHungry(): %u bytes never used\n",
                (unsigned)(hwm1 * sizeof(StackType_t)));

  stackHungry();

  UBaseType_t hwm2 = uxTaskGetStackHighWaterMark(NULL);
  Serial.printf("After stackHungry(): %u bytes never used\n",
                (unsigned)(hwm2 * sizeof(StackType_t)));

  Serial.printf("Approximate peak stack used: %u bytes\n",
                (unsigned)((4096 / sizeof(StackType_t) - hwm2)
                           * sizeof(StackType_t)));

  // Part D: Fixed-size memory pool
  Serial.println("\n--- PART D: MEMORY POOL ---");

  void *blk[NUM_BLOCKS];

  for (int i = 0; i < NUM_BLOCKS; i++) {
    blk[i] = poolAlloc();
    Serial.printf("Block %d allocated at %p\n", i + 1, blk[i]);
  }

  void *extra = poolAlloc();

  if (extra == NULL) {
    Serial.println("Block 5: FAILED - pool FULL");
  }

  poolFree(blk[1]);
  Serial.println("Block 2 released");

  void *again = poolAlloc();

  Serial.printf("New request got %p (%s)\n", again,
                again == blk[1] ? "same block reused" : "other block");

  for (int i = 0; i < NUM_BLOCKS; i++) {
    poolFree(blk[i]);
  }

  poolFree(again);
  Serial.println("All blocks released");

  // Part E: Pool versus heap timing
  Serial.println("\n--- PART E: SPEED TEST ---");

  uint32_t t0 = micros();

  for (int i = 0; i < 1000; i++) {
    void *p = poolAlloc();
    poolFree(p);
  }

  uint32_t tPool = micros() - t0;

  t0 = micros();

  for (int i = 0; i < 1000; i++) {
    void *p = pvPortMalloc(BLOCK_SIZE);
    if (p != NULL) {
      vPortFree(p);
    }
  }

  uint32_t tHeap = micros() - t0;

  Serial.printf("Memory pool: %lu us total\n",
                (unsigned long)tPool);
  Serial.printf("Heap: %lu us total\n",
                (unsigned long)tHeap);

  Serial.println("\n--- MEMORY TEST COMPLETE ---");

  vTaskDelete(NULL);
}

void setup() {
  Serial.begin(115200);
  delay(1000);

  Serial.println("==============================");
  Serial.println("RTOS MEMORY MANAGEMENT TEST");
  Serial.println("==============================");

  // ESP32 task stack size is specified in bytes
  xTaskCreate(MemoryTask, "MemoryTask", 4096, NULL, 1, NULL);
}

void loop() {
  vTaskDelay(pdMS_TO_TICKS(1000));
}
